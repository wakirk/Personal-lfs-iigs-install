install_GSOS_p11kit0() {
    packsearch $1
    local        package="$results_display_name"  # Display Name
    local   package_name="$results_package_name"  # Data Name
    local    tar_package="$results_tar_file"      # Tar package
    local working_folder="$results_working_dir"   # Working folder
    local        DL_FILE="$results_url"           # Download URL
    echoR                "Package $package_name"

    install_check "$package"
    [ $? -eq 2 ] && return 0;  # Already Installed.

    clear
    echo "Download (HTTP): https://github.com/p11-glue/p11-kit/releases/download/0.25.5/p11-kit-0.25.5.tar.xz"
    echo "Download MD5 sum: e9c5675508fcd8be54aa4c8cb8e794fc"
    echo "Download size: 980 KB"
    echo "Estimated disk space required: 94 MB (with tests)"
    echo "Estimated build time: 0.7 SBU (with tests)"

    #### Download and verify package ####

    cd /root/src
    echoL "Downloading $package..."
        lfs_download "$package" "$tar_package" "$DL_FILE"
        [ $? -ne 0 ] && exit 99

    #### Build and install ####

    echoR "Installing $package"

    echoL "Unpacking $package..."
        cd /root/src
        rm -vfR "$working_folder"    # clean up any previous work.
        tar -vxsf "$tar_package"
        cd "$working_folder"

    echoL "Configuring $package..."
        sed '20,$ d' -i trust/trust-extract-compat
        cat >> trust/trust-extract-compat << "EOF"
# Copy existing anchor modifications to /etc/ssl/local
/usr/libexec/make-ca/copy-trust-modifications

# Update trust stores
/usr/sbin/make-ca -r
EOF
        mkdir p11-build
        cd    p11-build
        meson setup ..            \
              --prefix=/usr       \
              --buildtype=release \
               -D trust_paths=/etc/pki/anchors

    echoL "Building $package..."
        ninja

    if [ "$SKIP_TESTS" = false ]; then
        echoL "Testing $package..."
            LC_ALL=C ninja test
        [ "$TESTS_WAIT" = true ] && read -p "Press Enter to Continue"
    fi

    echoL "Installing $package..."
        ninja install
        ln -sfv /usr/libexec/p11-kit/trust-extract-compat \
                /usr/bin/update-ca-certificates
        ln -sfv ./pkcs11/p11-kit-trust.so /usr/lib/libnssckbi.so

    #### Cleanup and Stamp ####

    echoL "Cleaning up..."
    lfs_cleanup "$package" "$working_folder"
    $INSTALLER "$package" Status Installed
    [ $? -ne 0 ] && exit 98
    return 0
}
